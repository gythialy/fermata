状态切换
Transition{Enter, Exit} + Motion{Collapse}

// 外壳：系统画框，内容只画墨 a.popover = mygo.NewFlyout(mygo.FlyoutOptions{ Tray: a.tray.Native(), Placement: mygo.PlacementBottom, Gap: 4, Width: 320, FitContent: true, MinHeight: 220, Popover: runtime.GOOS == "darwin", // 材质·箭头·开合 Shadow: true, CornerRadius: 14, Content: ui.View(a.popoverView), }) func (a *App) popoverView(c *ui.Context) { t := theme.Apply(c) ui.Column(c).Padding(t.Space(4)).Gap(t.Space(3)).Children(func() { ui.Row(c).Gap(t.Space(2)).Children(func() { ui.Icon(c, mark) // cool-moon crescent ui.Text(c, state).Bold() ui.Spacer(c) ui.Button(c, "").Children(func() { ui.Icon(c, more) }) }) ui.Text(c, label).Font(theme.Mono). FontSize(theme.Px(c, 34)).Bold(). ContentTransition(ui.ContentTransition{Numeric: true, Down: true}) ruler(c, t, share) // 60 根刻度 + 游标 ui.Text(c, detail).TextColor(t.TextMuted).MaxLines(2) ui.Spacer(c) ui.PrimaryButton(c, take).Grow(1) ui.Button(c, pause).Grow(1) ui.Divider(c) ui.Button(c, settings).Grow(1) }) }
01
高度不写死：FitContent 让 336 / 207 / 236 / 147 自己长出来，多一行动作也不会被裁。
02
刻度尺是唯一自绘的读数：60 根 1 px 发丝 + 游标，成本 61 个图元；角色与 Label 要自己补，别让它变成纯装饰。
03
主按钮要靠强调色站住：把 t.InactiveAccent 设为 t.Accent，三端都是朱红实心；不设的话 MyGo 会把非活动窗口的 PrimaryButton 画成普通按钮，与次级同貌。
04
数字用 ContentTransition：每 30 秒变一次，滚动读起来像钟，不像重绘。
